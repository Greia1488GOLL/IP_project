class FavoritesRepository:
    def __init__(self, db_path: Path) -> None:
        self.db_path = db_path
        self._initialize()

    def _connect(self) -> sqlite3.Connection:
        return sqlite3.connect(self.db_path)

    def _initialize(self) -> None:
        with closing(self._connect()) as connection:
            connection.execute(
                """
                CREATE TABLE IF NOT EXISTS favorites (
                    user_id INTEGER NOT NULL,
                    ticker TEXT NOT NULL,
                    PRIMARY KEY (user_id, ticker)
                )
                """
            )
            connection.commit()

    def add(self, user_id: int, ticker: str) -> bool:
        with closing(self._connect()) as connection:
            cursor = connection.execute(
                "INSERT OR IGNORE INTO favorites (user_id, ticker) VALUES (?, ?)",
                (user_id, ticker),
            )
            connection.commit()
            return cursor.rowcount > 0

    def remove(self, user_id: int, ticker: str) -> bool:
        with closing(self._connect()) as connection:
            cursor = connection.execute(
                "DELETE FROM favorites WHERE user_id = ? AND ticker = ?",
                (user_id, ticker),
            )
            connection.commit()
            return cursor.rowcount > 0

    def list_for_user(self, user_id: int) -> list[str]:
        with closing(self._connect()) as connection:
            rows = connection.execute(
                "SELECT ticker FROM favorites WHERE user_id = ? ORDER BY ticker",
                (user_id,),
            ).fetchall()
            return [row[0] for row in rows]

def normalize_ticker(raw_ticker: str) -> str:
    return raw_ticker.strip().upper()


def format_change(change: float, change_percent: float) -> str:
    arrow = "📈" if change >= 0 else "📉"
    return f"{arrow} {change:+.2f} ({change_percent:+.2f}%)"


def format_number(value: int | float | None) -> str:
    if value is None:
        return "Нет данных"
    if abs(value) >= 1_000_000_000_000:
        return f"{value / 1_000_000_000_000:.2f}T"
    if abs(value) >= 1_000_000_000:
        return f"{value / 1_000_000_000:.2f}B"
    if abs(value) >= 1_000_000:
        return f"{value / 1_000_000:.2f}M"
    return f"{value:,.2f}".replace(",", " ")


def escape_text(value: object) -> str:
    return html.escape(str(value))


def is_rate_limit_error(error: Exception) -> bool:
    message = str(error).lower()
    return "too many requests" in message or "rate limited" in message or "429" in message


def build_rate_limit_message() -> str:
    return (

        "Подождите 30-60 секунд и попробуйте снова."

def get_user_error_message(error: Exception) -> str:
    if is_rate_limit_error(error):
        return build_rate_limit_message()
    return f"Ошибка при получении данных: {error}"


def get_cached_stock_data(ticker: str) -> tuple[dict, dict] | None:
    cached = stock_data_cache.get(ticker)
    if not cached:
        return None

    cached_at, payload = cached
    if time.time() - cached_at > CACHE_TTL_SECONDS:
        stock_data_cache.pop(ticker, None)
        return None

    return payload


def set_cached_stock_data(ticker: str, fast_info: dict, info: dict) -> None:
    stock_data_cache[ticker] = (time.time(), (dict(fast_info), dict(info)))


def fetch_stock_data(ticker: str) -> tuple[dict, dict]:
    cached = get_cached_stock_data(ticker)
    if cached is not None:
        return cached

    stock = yf.Ticker(ticker)
    fast_info = dict(stock.fast_info)
    info = dict(stock.info)
    set_cached_stock_data(ticker, fast_info, info)
    return fast_info, info


def get_stock_snapshot(ticker: str) -> dict[str, str | float]:
    fast_info, info = fetch_stock_data(ticker)

    current_price = fast_info.get("lastPrice")
    previous_close = fast_info.get("previousClose")
    currency = fast_info.get("currency") or info.get("currency") or "USD"
    exchange = fast_info.get("exchange") or info.get("exchange") or "Unknown exchange"
    name = info.get("shortName") or info.get("longName") or ticker

    if current_price is None or previous_close is None:
        raise ValueError("Не удалось получить котировки. Проверьте тикер.")

    change = current_price - previous_close
    change_percent = (change / previous_close) * 100 if previous_close else 0

    return {
        "ticker": ticker,
        "name": name,
        "price": current_price,
        "previous_close": previous_close,
        "change": change,
        "change_percent": change_percent,
        "currency": currency,
        "exchange": exchange,
    }


def get_stock_profile(ticker: str) -> dict[str, str | float | None]:
    fast_info, info = fetch_stock_data(ticker)

    current_price = fast_info.get("lastPrice")
    previous_close = fast_info.get("previousClose")
    if current_price is None or previous_close is None:
        raise ValueError("Не удалось получить данные по акции. Проверьте тикер.")

    change = current_price - previous_close
    change_percent = (change / previous_close) * 100 if previous_close else 0

    return {
        "ticker": ticker,
        "name": info.get("shortName") or info.get("longName") or ticker,
        "price": current_price,
        "currency": fast_info.get("currency") or info.get("currency") or "USD",
        "exchange": fast_info.get("exchange") or info.get("exchange") or "Unknown exchange",
        "change": change,
        "change_percent": change_percent,
        "sector": info.get("sectorDisp") or info.get("sector"),
        "industry": info.get("industryDisp") or info.get("industry"),
        "country": info.get("country"),
        "market_cap": info.get("marketCap"),
        "fifty_two_week_low": info.get("fiftyTwoWeekLow"),
        "fifty_two_week_high": info.get("fiftyTwoWeekHigh"),
    }


def build_stock_message(snapshot: dict[str, str | float]) -> str:
    return (
        f"<b>{escape_text(snapshot['name'])}</b> (<code>{escape_text(snapshot['ticker'])}</code>)\n"
        f"Цена: <b>{snapshot['price']:.2f} {escape_text(snapshot['currency'])}</b>\n"
        f"Изменение за день: {format_change(snapshot['change'], snapshot['change_percent'])}\n"
        f"Предыдущее закрытие: {snapshot['previous_close']:.2f} {escape_text(snapshot['currency'])}\n"
        f"Биржа: {escape_text(snapshot['exchange'])}"
    )


def build_profile_message(profile: dict[str, str | float | None]) -> str:
    year_range = (
        f"{profile['fifty_two_week_low']:.2f} - {profile['fifty_two_week_high']:.2f}"
        if profile["fifty_two_week_low"] is not None and profile["fifty_two_week_high"] is not None
        else "Нет данных"
    )

    return (
        f"<b>{escape_text(profile['name'])}</b> (<code>{escape_text(profile['ticker'])}</code>)\n"
        f"Цена: <b>{profile['price']:.2f} {escape_text(profile['currency'])}</b>\n"
        f"Изменение за день: {format_change(profile['change'], profile['change_percent'])}\n"
        f"Сектор: {escape_text(profile['sector'] or 'Нет данных')}\n"
        f"Отрасль: {escape_text(profile['industry'] or 'Нет данных')}\n"
        f"Страна: {escape_text(profile['country'] or 'Нет данных')}\n"
        f"Капитализация: {escape_text(format_number(profile['market_cap']))}\n"
        f"Диапазон 52 недель: {escape_text(year_range)}\n"
        f"Биржа: {escape_text(profile['exchange'])}"
    )


def build_compare_message(
    first_snapshot: dict[str, str | float],
    second_snapshot: dict[str, str | float],
) -> str:
    better = first_snapshot if first_snapshot["change_percent"] >= second_snapshot["change_percent"] else second_snapshot

    return (
        "<b>Сравнение акций</b>\n\n"
        f"<b>{escape_text(first_snapshot['ticker'])}</b>: {first_snapshot['price']:.2f} "
        f"{escape_text(first_snapshot['currency'])} | {format_change(first_snapshot['change'], first_snapshot['change_percent'])}\n"
        f"<b>{escape_text(second_snapshot['ticker'])}</b>: {second_snapshot['price']:.2f} "
        f"{escape_text(second_snapshot['currency'])} | {format_change(second_snapshot['change'], second_snapshot['change_percent'])}\n\n"
        f"Лидер за день: <code>{escape_text(better['ticker'])}</code>"
    )

def set_state(context: ContextTypes.DEFAULT_TYPE, state: str) -> None:
    current_state = context.user_data.get("state", STATE_MAIN_MENU)
    if current_state != state:
        history = context.user_data.setdefault("history", [])
        history.append(current_state)
    context.user_data["state"] = state


def reset_to_menu(context: ContextTypes.DEFAULT_TYPE) -> None:
    context.user_data["state"] = STATE_MAIN_MENU
    context.user_data["history"] = []


def pop_previous_state(context: ContextTypes.DEFAULT_TYPE) -> str:
    history = context.user_data.setdefault("history", [])
    previous_state = history.pop() if history else STATE_MAIN_MENU
    context.user_data["state"] = previous_state
    return previous_state

async def price_command(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    if not context.args:
        set_state(context, STATE_WAIT_PRICE)
        await update.message.reply_text(
            "Введите тикер акции. Например: AAPL",
            reply_markup=back_keyboard(),
        )
        return

    ticker = normalize_ticker(context.args[0])
    try:
        await show_price(update, ticker)
    except Exception as error:
        logger.exception("Failed to fetch stock data for %s", ticker)
        await update.message.reply_text(get_user_error_message(error))

        logger.exception("Failed to compare %s and %s", first_ticker, second_ticker)
        await update.message.reply_text(get_user_error_message(error))

async def favorites_command(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    user_id = update.effective_user.id
    tickers = favorites_repository.list_for_user(user_id)

    if not tickers:
        await update.message.reply_text("Список избранного пока пуст.")
        return

    messages: list[str] = []
    for ticker in tickers:
        try:
            snapshot = get_stock_snapshot(ticker)
            messages.append(build_stock_message(snapshot))
        except Exception as error:
            logger.exception("Failed to fetch stock data for favorite %s", ticker)
            messages.append(
                f"<b>{escape_text(ticker)}</b>\n{escape_text(get_user_error_message(error))}"
            )

    await update.message.reply_text("\n\n".join(messages), parse_mode="HTML")

            "Введите тикер, чтобы получить информацию о компании. Например: AAPL",
            reply_markup=back_keyboard(),
        )
        return

    if previous_state == STATE_WAIT_COMPARE:
        await update.message.reply_text(
            "Введите два тикера через пробел. Например: AAPL MSFT",
            reply_markup=back_keyboard(),
        )
        return

    if previous_state == STATE_WAIT_ADD_FAVORITE:
        await update.message.reply_text(
            "Введите тикер, который хотите добавить в избранное. Например: TSLA",
            reply_markup=back_keyboard(),
        )
        return

    if previous_state == STATE_WAIT_REMOVE_FAVORITE:
        await update.message.reply_text(
            "Введите тикер, который хотите удалить из избранного. Например: TSLA",
            reply_markup=back_keyboard(),

    reset_to_menu(context)
    await send_main_menu(update, "Вернулись в главное меню.")


async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    if not update.message or not update.message.text:
        return

    text = update.message.text.strip()

    if text == BUTTON_PRICE:
        set_state(context, STATE_WAIT_PRICE)
        await update.message.reply_text(
            "Введите тикер акции. Например: AAPL",
            reply_markup=back_keyboard(),
        )
        return

    if text == BUTTON_INFO:
        set_state(context, STATE_WAIT_INFO)
        await update.message.reply_text(
            "Введите тикер, чтобы получить информацию о компании. Например: AAPL",
            reply_markup=back_keyboard(),
        )
        return

    if text == BUTTON_COMPARE:
        set_state(context, STATE_WAIT_COMPARE)
        await update.message.reply_text(
            "Введите два тикера через пробел. Например: AAPL MSFT",
            reply_markup=back_keyboard(),
        )
        return

    if text == BUTTON_ADD_FAVORITE:
        set_state(context, STATE_WAIT_ADD_FAVORITE)
        await update.message.reply_text(
            "Введите тикер, который хотите добавить в избранное. Например: TSLA",
            reply_markup=back_keyboard(),
        )
        return

    if text == BUTTON_REMOVE_FAVORITE:
        set_state(context, STATE_WAIT_REMOVE_FAVORITE)
        await update.message.reply_text(
            "Введите тикер, который хотите удалить из избранного. Например: TSLA",
            reply_markup=back_keyboard(),
        )
        return

    if text == BUTTON_FAVORITES:
        await favorites_command(update, context)
        return

    if text == BUTTON_BACK:
        await back_command(update, context)
        return

    state = context.user_data.get("state", STATE_MAIN_MENU)

    if state == STATE_WAIT_PRICE:
        ticker = normalize_ticker(text)
        try:
            await show_price(update, ticker)
            reset_to_menu(context)
            await send_main_menu(update)
        except Exception as error:
            logger.exception("Failed to fetch stock data for %s", ticker)
            await update.message.reply_text(
                f"{get_user_error_message(error)}\nПопробуйте другой тикер или нажмите Назад.",
                reply_markup=back_keyboard(),
            )
        return

    if state == STATE_WAIT_INFO:
        ticker = normalize_ticker(text)
        try:
            await show_info(update, ticker)
            reset_to_menu(context)
            await send_main_menu(update)
        except Exception as error:
            logger.exception("Failed to fetch company info for %s", ticker)
            await update.message.reply_text(
                f"{get_user_error_message(error)}\nПопробуйте другой тикер или нажмите Назад.",
                reply_markup=back_keyboard(),
            )
        return

    if state == STATE_WAIT_COMPARE:
        parts = text.split()
        if len(parts) < 2:
            await update.message.reply_text(
                "Нужно ввести два тикера через пробел. Например: AAPL MSFT",
                reply_markup=back_keyboard(),
            )
            return
        first_ticker, second_ticker = normalize_ticker(parts[0]), normalize_ticker(parts[1])
        try:
            await show_compare(update, first_ticker, second_ticker)
            reset_to_menu(context)
            await send_main_menu(update)
        except Exception as error:
            logger.exception("Failed to compare %s and %s", first_ticker, second_ticker)
            await update.message.reply_text(
                f"{get_user_error_message(error)}\nПопробуйте другие тикеры или нажмите Назад.",
                reply_markup=back_keyboard(),
            )
        return

    if state == STATE_WAIT_ADD_FAVORITE:
        await add_favorite_by_ticker(update, normalize_ticker(text))
        reset_to_menu(context)
        await send_main_menu(update)
        return

    if state == STATE_WAIT_REMOVE_FAVORITE:
        await remove_favorite_by_ticker(update, normalize_ticker(text))
        reset_to_menu(context)
        await send_main_menu(update)
        return

    await send_main_menu(update, "Не понял команду. Выберите действие кнопкой ниже.")


def main() -> None:
    token = os.getenv("TELEGRAM_BOT_TOKEN")
    if not token:
        raise RuntimeError("Не найден TELEGRAM_BOT_TOKEN. Добавьте его в .env файл.")

    application = Application.builder().token(token).build()

    application.add_handler(CommandHandler("start", start_command))
    application.add_handler(CommandHandler("help", help_command))
    application.add_handler(CommandHandler("price", price_command))
    application.add_handler(CommandHandler("info", info_command))
    application.add_handler(CommandHandler("compare", compare_command))
